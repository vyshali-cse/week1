<h2>Registration Form</h2>

<form onsubmit="return validate()">
Name: <input type="text" id="name"><br><br>
Email: <input type="text" id="email"><br><br>
Password: <input type="password" id="pass"><br><br>

<button type="submit">Register</button>
<p id="msg"></p>
</form>

<script>
function validate() {
 var n = document.getElementById("name").value;
 var e = document.getElementById("email").value;
 var p = document.getElementById("pass").value;

 if(n=="" || e=="" || p=="") {
   document.getElementById("msg").innerHTML="All fields are required";
   return false;
 }
 if(!e.includes("@")) {
   document.getElementById("msg").innerHTML="Invalid Email";
   return false;
 }
 if(p.length<6) {
   document.getElementById("msg").innerHTML="Password must be 6 characters";
   return false;
 }

 document.getElementById("msg").innerHTML="Registration Successful";
 return false;
}
</script>
